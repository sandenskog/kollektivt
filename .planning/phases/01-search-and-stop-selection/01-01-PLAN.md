   ```typescript
   // SL Transport API site shape
   export interface SLSite {
     id: number;
     name: string;
     lat: number;
     lon: number;
     stop_areas?: { id: number; name: string }[];
   }

   // Nominatim search result
   export interface NominatimResult {
     display_name: string;
     lat: string;
     lon: string;
   }

   // A stop with computed distance (from findNearbyStops)
   export interface NearbyStop extends SLSite {
     distance: number; // meters
   }

   // What gets persisted to localStorage
   export interface SelectedStop {
     id: number;
     name: string;
     lat: number;
     lon: number;
   }

   // Versioned localStorage schema
   export interface PersistedData {
     version: number;
     selectedStops: SelectedStop[];
   }
   ```

  <action>
1. Install vitest for testing: `npm install -D vitest` and add `"test": "vitest run"` to package.json scripts.

2. Create `src/services/distance.ts`:
   - `haversineDistance(lat1, lon1, lat2, lon2): number` — returns distance in meters
   - `findNearbyStops(lat, lon, sites, maxDistance=1000, maxResults=10): NearbyStop[]` — filters, sorts by distance, slices
   - Import types from `src/types/index.ts`

3. Create `src/services/distance.test.ts` with tests per the behavior block above. Run tests (expect RED first for TDD, then GREEN).

4. Create `src/services/sl-sites.ts`:
   - `loadSites(): Promise<SLSite[]>` — fetches `https://transport.integration.sl.se/v1/sites?expand=true`, caches in module-level variable, returns cached on subsequent calls
   - Include error handling: throw descriptive error on non-OK response

5. Create `src/services/nominatim.ts`:
   - `searchAddress(query: string, signal?: AbortSignal): Promise<NominatimResult[]>` — calls Nominatim `/search` with params: format=jsonv2, countrycodes=se, limit=5, accept-language=sv
   - MUST include User-Agent header: `Kollektivt/1.0 (kollektivt.sandenskog.se)`
   - Returns parsed NominatimResult array

6. Create `src/hooks/usePersistedStops.ts`:
   - `usePersistedStops()` — returns `{ selectedStops, toggleStop, isSelected }`
   - Reads from localStorage on mount using key `kollektivt_data`
   - Schema version = 1, includes `migrate()` function for future upgrades
   - `toggleStop(stop: SelectedStop)` adds if not present, removes if present, saves immediately
   - `isSelected(id: number)` returns boolean

7. Create `src/hooks/useNominatim.ts`:
   - `useNominatim(query: string, debounceMs=300)` — returns `{ results, loading }`
   - Debounces via setTimeout in useEffect, aborts previous request via AbortController
   - Minimum 3 characters before searching, returns empty array for shorter queries
   - Calls `searchAddress` from the nominatim service

8. Create `src/hooks/useGeolocation.ts`:
   - `useGeolocation()` — returns `{ lat, lon, loading, error, requestLocation }`
   - `requestLocation` triggers `navigator.geolocation.getCurrentPosition`
   - Handles all error codes with user-friendly messages in English (UI text)
   - enableHighAccuracy=false, timeout=10000

9. Create `src/hooks/useNearbyStops.ts`:
   - `useNearbyStops(lat: number | null, lon: number | null)` — returns `{ stops, loading, error }`
   - When lat/lon change (and are non-null), calls `loadSites()` then `findNearbyStops()`
   - Returns up to 10 stops within 1km sorted by distance
