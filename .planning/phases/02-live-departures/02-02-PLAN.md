export interface SelectedStop {
  id: number;

  lon: number;
}

export interface Departure {
  destination: string;
  direction_code: number;
  direction: string;
  state: 'EXPECTED' | 'ATSTOP' | 'CANCELLED';
  display: string;
  scheduled: string;
  expected: string;
  journey: { id: number; state: string; prediction_state: string };
  stop_area: { id: number; name: string; type: string };
  stop_point: { id: number; name: string; designation: string };
  line: {
    id: number;
    designation: string;
    transport_mode: 'BUS' | 'METRO' | 'TRAIN' | 'TRAM';
    group_of_lines: string;
  };
  deviations: Array<{
    importance_level: number;
    consequence: string;
    message: string;
  }>;
}

export interface StopDeviation {
  id: number;
  importance_level: number;
  message: string;
  scope: {
    stop_areas: Array<{ id: number; name: string }>;
    lines: Array<{ id: number; designation: string; transport_mode: string }>;
  };
}

export interface DepartureResponse {
  departures: Departure[];
  stop_deviations: StopDeviation[];
}

export interface StopFilters {
  directions?: number[];
  lines?: number[];
}

  <action>
    Build four components bottom-up:

    1. **DepartureRow.tsx** -- Single departure row:
       - Props: `departure: Departure`, `tick: number` (tick triggers re-render for countdown refresh)
       - Compute countdown using `getCountdown(departure)` and absolute time using `getAbsoluteTime(departure)`
       - Transport mode badge: text badge with CSS class per mode. Use single-letter abbreviations: T (METRO), B (BUS), J (TRAIN), S (TRAM). Render as `<span className="mode-badge mode-{mode.toLowerCase()}">T</span>`
       - Layout per row: [mode-badge] [line-designation] [destination] [countdown] [absolute-time]
       - Delay handling: if `getDelayMinutes(departure) > 0`, show scheduled time with strikethrough `<s>` and expected time beside it, plus "+{delay} min" text with a CSS class `delay-indicator`
       - Cancelled handling: if `departure.state === 'CANCELLED'`, show entire row with class `cancelled`, destination text with strikethrough, and "Cancelled" label instead of countdown
       - Past departures: if countdown computes to <= 0 min AND state is not ATSTOP, hide the row (return null). If state is ATSTOP, show "Now"

    2. **DepartureFilters.tsx** -- Direction and line filter chips:
       - Props: `departures: Departure[]` (all unfiltered departures for this stop, to extract available options), `filters: StopFilters`, `onFiltersChange: (filters: StopFilters) => void`
       - Extract unique directions: `Array.from(new Set(departures.map(d => d.direction_code)))` with their text labels from `d.direction`
       - Extract unique lines: `Array.from(new Set(departures.map(d => d.line.id)))` with their designations from `d.line.designation`
       - Render as two rows of toggle chips: "Directions:" row and "Lines:" row
       - A chip is "active" (included) by default. Clicking toggles it off (excluded). When ALL chips are active, show no special state (this is the default "show everything" mode)
       - When filters.directions is undefined or empty array, ALL directions are shown. When it has values, only those direction_codes are shown.
       - Same logic for filters.lines with line IDs
       - Only render the filter section if there are 2+ directions or 2+ lines (no point filtering if only one option)

    3. **StopDepartures.tsx** -- One stop's departure section:
       - Props: `stop: SelectedStop`, `response: DepartureResponse | undefined`, `error: boolean`, `filters: StopFilters`, `onFiltersChange: (filters: StopFilters) => void`, `tick: number`
       - Render stop name as `<h3>` header
       - If response is undefined (loading): show "Loading departures..." text
       - If error is true AND response exists: show discrete "Could not update" message below header (keep showing stale data)
       - If error is true AND response is undefined: show "Could not load departures" error message
       - Disruption banner: if `response.stop_deviations.length > 0`, render a warning banner div with class `disruption-banner` listing each deviation message. Also check departure-level deviations and collect unique messages.
       - Apply client-side filtering: filter `response.departures` by `filters.directions` (if set, keep only matching direction_code) and `filters.lines` (if set, keep only matching line.id)
       - Sort filtered departures by expected time (nearest first)
       - Slice to first 5 departures
       - Render DepartureFilters (pass all unfiltered departures for option extraction)
       - Render each departure as DepartureRow

    4. **DepartureList.tsx** -- Container for all stop departure sections:
       - Props: `stops: SelectedStop[]`, `departuresByStop: Map<number, DepartureResponse>`, `errors: Map<number, boolean>`, `filters: Record<number, StopFilters>`, `onFiltersChange: (stopId: number, filters: StopFilters) => void`, `refresh: () => void`, `tick: number`
       - If stops.length === 0: return null
       - Render a section with header "Departures" and a manual refresh button (discrete refresh icon, use unicode refresh symbol or similar)
       - Map over stops, rendering StopDepartures for each with the correct data from maps
       - Pass `errors.get(stop.id) ?? false` and `departuresByStop.get(stop.id)` to each
